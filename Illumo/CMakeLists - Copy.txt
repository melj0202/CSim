if(MSVC)
    # Disables C++ exceptions entirely in MSVC
    target_compile_options(Illumo PRIVATE /EHs- /EHc-)
    # /wd4577: Disables the 'noexcept used with no exception handling' warning
    # /wd5045: Disables the Spectre mitigation warning (good to pair with this)
    target_compile_options(Illumo PRIVATE /wd4577 /wd5045)
    
    # Explicitly tells the MSVC standard library that exceptions are off
    target_compile_definitions(Illumo PRIVATE _HAS_EXCEPTIONS=0)
else()
    # Disables exceptions in GCC and Clang
    target_compile_options(Illumo PRIVATE -fno-exceptions)
    target_compile_options(Illumo PRIVATE -Wno-unknown-pragmas)
endif()
